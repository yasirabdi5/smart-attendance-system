========================================================= # SFace + YuNet # ========================================================= SFACE_MODEL_PATH = ( "face_data/models/face_recognition_sface_2021dec.onnx" ) YUNET_MODEL_PATH = ( "face_data/models/face_detection_yunet_2023mar.onnx" ) sface_recognizer = cv2.FaceRecognizerSF_create( SFACE_MODEL_PATH, "" ) yunet_detector = cv2.FaceDetectorYN_create( YUNET_MODEL_PATH, "", (320, 320), 0.9, 0.3, 5000 ) def get_face_feature(image_path): image = cv2.imread(image_path) if image is None: return None height, width = image.shape[:2] yunet_detector.setInputSize( (width, height) ) _, faces = yunet_detector.detect(image) if faces is None: return None face = max( faces, key=lambda f: f[2] * f[3] ) aligned_face = sface_recognizer.alignCrop( image, face ) feature = sface_recognizer.feature( aligned_face ) return feature def compare_faces_sface( image1_path, image2_path ): feature1 = get_face_feature( image1_path ) feature2 = get_face_feature( image2_path ) if feature1 is None or feature2 is None: return None score = sface_recognizer.match( feature1, feature2, cv2.FaceRecognizerSF_FR_COSINE ) return float(score) # ========================================================= # FACE DETECTION # ========================================================= FACE_CASCADE_PATH = ( "face_data/haarcascade_frontalface_default.xml" ) face_detector = cv2.CascadeClassifier( FACE_CASCADE_PATH ) def detect_face(image_path): image = cv2.imread(image_path) if image is None: return None gray = cv2.cvtColor( image, cv2.COLOR_BGR2GRAY ) faces = face_detector.detectMultiScale( gray, scaleFactor=1.1, minNeighbors=5, minSize=(100, 100) ) if len(faces) == 0: return None x, y, w, h = max( faces, key=lambda face: face[2] * face[3] ) face = gray[ y:y+h, x:x+w ] return face #
